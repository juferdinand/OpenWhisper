cmake_minimum_required(VERSION 3.22)
project(OpenWhisperMacRetirementProbe LANGUAGES CXX)
if(NOT APPLE OR NOT CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "14.0")
  message(FATAL_ERROR "The retirement probe requires an explicit macOS 14 deployment target")
endif()
if(NOT EXISTS "${NODE_HEADERS}/node_api.h")
  message(FATAL_ERROR "Checksum-pinned Node-API headers are required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(openwhisper_macos_retirement_probe MODULE retirement.cpp)
target_include_directories(openwhisper_macos_retirement_probe PRIVATE "${NODE_HEADERS}")
target_compile_definitions(openwhisper_macos_retirement_probe PRIVATE NAPI_VERSION=8 OPENWHISPER_RETIREMENT_PROBE=1)
target_compile_options(openwhisper_macos_retirement_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(openwhisper_macos_retirement_probe PRIVATE proc)
target_link_options(openwhisper_macos_retirement_probe PRIVATE -undefined dynamic_lookup)
set_target_properties(openwhisper_macos_retirement_probe PROPERTIES PREFIX "" SUFFIX ".node")
