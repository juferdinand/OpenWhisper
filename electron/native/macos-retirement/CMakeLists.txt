cmake_minimum_required(VERSION 3.22)
project(OpenWhisperMacRetirementProbe LANGUAGES CXX)
if(NOT APPLE OR NOT CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "14.0")
  message(FATAL_ERROR "The retirement probe requires an explicit macOS 14 deployment target")
endif()
if(NOT EXISTS "${NODE_HEADERS}/node_api.h")
  message(FATAL_ERROR "Checksum-pinned Node-API headers are required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(openwhisper_macos_retirement_probe MODULE retirement.cpp)
# The historical probe remains the sole default target. Production requires an
# explicit target selection and has a separate artifact/export role.
add_library(openwhisper_macos_retirement MODULE EXCLUDE_FROM_ALL retirement.cpp)
foreach(target openwhisper_macos_retirement_probe openwhisper_macos_retirement)
  target_include_directories(${target} PRIVATE "${NODE_HEADERS}")
  target_compile_definitions(${target} PRIVATE NAPI_VERSION=8)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(${target} PRIVATE proc)
  target_link_options(${target} PRIVATE -undefined dynamic_lookup)
  set_target_properties(${target} PROPERTIES PREFIX "" SUFFIX ".node")
endforeach()
target_compile_definitions(openwhisper_macos_retirement_probe PRIVATE OPENWHISPER_RETIREMENT_PROBE=1)
target_compile_definitions(openwhisper_macos_retirement PRIVATE OPENWHISPER_RETIREMENT_PRODUCTION=1)
