cmake_minimum_required(VERSION 3.22)
project(OpenWhisperCapture LANGUAGES C CXX)

# The initial native capture slice is Linux PulseAudio only, with a no-device synthetic path.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "CoreAudio and ALSA have separate implementation and acceptance gates")
endif()
if(NOT EXISTS "${MINIAUDIO_SOURCE}/miniaudio.h" OR NOT EXISTS "${NODE_HEADERS}/node_api.h")
  message(FATAL_ERROR "Checksum-pinned miniaudio and Node-API headers are required")
endif()
file(SHA256 "${MINIAUDIO_SOURCE}/miniaudio.h" MINIAUDIO_HASH)
if(NOT MINIAUDIO_HASH STREQUAL "ac7af4de748b7e26b777f37e01cee313a308a7296a3eb080e2906b320cc55c89")
  message(FATAL_ERROR "Miniaudio source checksum mismatch")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(openwhisper_capture MODULE capture.cpp miniaudio.c)
target_include_directories(openwhisper_capture PRIVATE "${NODE_HEADERS}" "${MINIAUDIO_SOURCE}")
target_compile_definitions(openwhisper_capture PRIVATE NAPI_VERSION=8 MA_ENABLE_ONLY_SPECIFIC_BACKENDS
  MA_ENABLE_PULSEAUDIO MA_NO_DECODING MA_NO_ENCODING MA_NO_RESOURCE_MANAGER MA_NO_NODE_GRAPH MA_NO_ENGINE)
target_compile_options(openwhisper_capture PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Werror>")
target_link_libraries(openwhisper_capture PRIVATE pthread dl m)
set_target_properties(openwhisper_capture PROPERTIES PREFIX "" SUFFIX ".node")
