cmake_minimum_required(VERSION 3.22)
project(OpenWhisperSpeech LANGUAGES C CXX)

if(NOT EXISTS "${WHISPER_SOURCE}/include/parakeet.h" OR NOT EXISTS "${NODE_HEADERS}/node_api.h")
  message(FATAL_ERROR "Checksum-pinned speech source and Node-API headers are required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(OPENWHISPER_BACKEND "cpu" CACHE STRING "Speech backend: cpu, vulkan or metal")
set_property(CACHE OPENWHISPER_BACKEND PROPERTY STRINGS cpu vulkan metal)
if(NOT OPENWHISPER_BACKEND MATCHES "^(cpu|vulkan|metal)$")
  message(FATAL_ERROR "Unsupported speech backend")
endif()
if(OPENWHISPER_BACKEND STREQUAL "vulkan" AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The Vulkan speech profile requires Linux")
endif()
if(OPENWHISPER_BACKEND STREQUAL "metal" AND NOT APPLE)
  message(FATAL_ERROR "The Metal speech profile requires macOS")
endif()
# CPU remains compiled in every profile; manual CPU choice is a per-model request.
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
if(OPENWHISPER_BACKEND STREQUAL "vulkan")
  set(GGML_VULKAN ON CACHE BOOL "" FORCE)
elseif(OPENWHISPER_BACKEND STREQUAL "metal")
  set(GGML_METAL ON CACHE BOOL "" FORCE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
endif()
foreach(flag SSE42 AVX AVX2 BMI2 FMA F16C)
  set(GGML_${flag} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${WHISPER_SOURCE}" speech EXCLUDE_FROM_ALL)
add_library(openwhisper_speech MODULE speech_bridge.cpp speech_binding.cpp)
target_include_directories(openwhisper_speech PRIVATE "${NODE_HEADERS}" "${WHISPER_SOURCE}/include" "${WHISPER_SOURCE}/ggml/include")
target_compile_definitions(openwhisper_speech PRIVATE NAPI_VERSION=8)
target_link_libraries(openwhisper_speech PRIVATE whisper parakeet)
set_target_properties(openwhisper_speech PROPERTIES PREFIX "" SUFFIX ".node")
if(APPLE)
  target_link_options(openwhisper_speech PRIVATE "-undefined" "dynamic_lookup")
endif()
