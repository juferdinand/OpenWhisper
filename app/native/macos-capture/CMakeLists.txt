cmake_minimum_required(VERSION 3.22)
project(OpenWhisperMacCapture LANGUAGES CXX OBJCXX)
if(NOT APPLE OR NOT CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "14.0")
  message(FATAL_ERROR "The macOS capture edge requires an explicit macOS 14 deployment target")
endif()
if(NOT EXISTS "${NODE_HEADERS}/node_api.h")
  message(FATAL_ERROR "Checksum-pinned Node-API headers are required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
add_library(openwhisper_macos_capture MODULE capture.mm reference.mm)
target_include_directories(openwhisper_macos_capture PRIVATE "${NODE_HEADERS}")
target_compile_definitions(openwhisper_macos_capture PRIVATE NAPI_VERSION=8)
target_compile_options(openwhisper_macos_capture PRIVATE -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(openwhisper_macos_capture PRIVATE "-framework Foundation" "-framework AVFoundation")
target_link_options(openwhisper_macos_capture PRIVATE -undefined dynamic_lookup)
set_target_properties(openwhisper_macos_capture PROPERTIES PREFIX "" SUFFIX ".node")
