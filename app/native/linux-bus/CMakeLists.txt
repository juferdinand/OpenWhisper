cmake_minimum_required(VERSION 3.22)
project(OpenWhisperLinuxBus LANGUAGES CXX)
if(NOT EXISTS "${NODE_HEADERS}/node_api.h")
  message(FATAL_ERROR "Checksum-pinned Node-API headers are required")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0 gio-unix-2.0)
add_library(openwhisper_linux_bus MODULE binding.cpp codec.cpp)
target_compile_features(openwhisper_linux_bus PRIVATE cxx_std_17)
target_compile_definitions(openwhisper_linux_bus PRIVATE NAPI_VERSION=8)
target_include_directories(openwhisper_linux_bus PRIVATE "${NODE_HEADERS}")
target_link_libraries(openwhisper_linux_bus PRIVATE PkgConfig::GIO)
target_compile_options(openwhisper_linux_bus PRIVATE -Wall -Wextra -Werror)
set_target_properties(openwhisper_linux_bus PROPERTIES PREFIX "" SUFFIX ".node")
